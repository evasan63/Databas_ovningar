-- Skapa tabell Avdelning
CREATE TABLE  testTrigger
(
    SID int NOT NULL,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL
    PRIMARY KEY (SID)
);
GO

----------------------------------------------------------------

-- Create trigger for INSERT
create trigger testTriggerInsert
on testTrigger
for insert
as
Print 'En ny post har lagds in'
select 'Ny post:', * from inserted
select * from testTrigger


-- Test trigger by inserting new value
INSERT INTO testTrigger (SID, FirstName, LastName)
VALUES (5, 'Kajsa', 'Warg');


-- RESULT --

-- Create trigger for UPDATE
create trigger testTriggerModify
on testTrigger
for update
as
Print 'En post har uppdaterats'
select 'Deleted: Den gamla datan var;', * from deleted
select 'Inserted: Den nya datan är;', * from inserted
select * from testTrigger

-- Test trigger for UPDATE
update testTrigger
set LastName = 'Persson'
where FirstName = 'Sture'
